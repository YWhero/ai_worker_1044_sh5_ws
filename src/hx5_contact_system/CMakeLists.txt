cmake_minimum_required(VERSION 3.16)
project(hx5_contact_system)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-transport13 REQUIRED)
add_library(hx5_contact_system SHARED src/contact_system.cpp)
target_compile_features(hx5_contact_system PUBLIC cxx_std_17)
target_link_libraries(hx5_contact_system gz-sim8::gz-sim8 gz-transport13::gz-transport13)
install(TARGETS hx5_contact_system DESTINATION lib)
ament_package()
